-- 1. BỔ SUNG SNAPSHOT TÊN ĐỊA ĐIỂM CHO ĐƠN HÀNG
-- ============================================================

ALTER TABLE nv_don_hang
ADD COLUMN IF NOT EXISTS
    ten_dia_diem_nhan_snapshot VARCHAR(255);


-- Lưu lại tên địa điểm cũ trước khi xóa bảng.
UPDATE nv_don_hang dh
SET ten_dia_diem_nhan_snapshot =
    COALESCE(
        dd.ten_dia_diem,
        'Địa điểm khác'
    )
FROM dm_dia_diem_nhan_hang dd
WHERE dh.dia_diem_nhan_id = dd.id
    AND dh.ten_dia_diem_nhan_snapshot IS NULL;


-- Những đơn không chọn địa điểm từ danh mục.
UPDATE nv_don_hang
SET ten_dia_diem_nhan_snapshot =
    'Địa điểm khác'
WHERE ten_dia_diem_nhan_snapshot IS NULL;


ALTER TABLE nv_don_hang
ALTER COLUMN ten_dia_diem_nhan_snapshot
SET NOT NULL;


-- ============================================================
-- 2. XÓA LIÊN KẾT TỚI ĐỊA ĐIỂM CŨ
-- ============================================================

ALTER TABLE nv_don_hang
DROP CONSTRAINT IF EXISTS
    nv_don_hang_dia_diem_nhan_id_fkey;


-- ID địa điểm cũ sẽ không còn giá trị sau khi tạo lại bảng.
UPDATE nv_don_hang
SET dia_diem_nhan_id = NULL
WHERE dia_diem_nhan_id IS NOT NULL;


-- ============================================================
-- 3. XÓA BẢNG ĐỊA ĐIỂM CŨ
-- ============================================================

DROP TABLE IF EXISTS
    dm_dia_diem_nhan_hang;


-- ============================================================
-- 4. TẠO LẠI BẢNG ĐỊA ĐIỂM THEO NHÂN VIÊN
-- ============================================================

CREATE TABLE dm_dia_diem_nhan_hang (
    id BIGSERIAL PRIMARY KEY,

    nhan_vien_id INTEGER NOT NULL,

    ma_dia_diem VARCHAR(50) NOT NULL,

    ten_dia_diem VARCHAR(255) NOT NULL,

    dia_chi_chi_tiet VARCHAR(500) NOT NULL,

    loai_dia_diem INTEGER DEFAULT 20 NOT NULL,

    la_mac_dinh BOOLEAN DEFAULT FALSE NOT NULL,

    ghi_chu VARCHAR(500),

    thu_tu_hien_thi INTEGER DEFAULT 0 NOT NULL,

    active BOOLEAN DEFAULT TRUE NOT NULL,

    created_at TIMESTAMP DEFAULT NOW() NOT NULL,

    updated_at TIMESTAMP DEFAULT NOW() NOT NULL,

    CONSTRAINT fk_dm_dia_diem_nhan_hang_nhan_vien
        FOREIGN KEY (
            nhan_vien_id
        )
        REFERENCES dm_nhan_vien(id),

    CONSTRAINT uq_dm_dia_diem_nhan_hang
        UNIQUE (
            nhan_vien_id,
            ma_dia_diem
        ),

    CONSTRAINT chk_dm_dia_diem_nhan_hang_loai
        CHECK (
            loai_dia_diem IN (
                10,
                20
            )
        ),

    CONSTRAINT chk_dm_dia_diem_nhan_hang_thu_tu
        CHECK (
            thu_tu_hien_thi >= 0
        )
);


COMMENT ON TABLE dm_dia_diem_nhan_hang IS
'Sổ địa điểm nhận hàng được lưu riêng theo từng nhân viên';


COMMENT ON COLUMN dm_dia_diem_nhan_hang.loai_dia_diem IS
'10=Địa điểm cố định, 20=Địa điểm ghi nhớ';


COMMENT ON COLUMN dm_dia_diem_nhan_hang.la_mac_dinh IS
'TRUE=Địa điểm được tự động chọn khi nhân viên đặt hàng';


-- ============================================================
-- 5. TẠO INDEX
-- ============================================================

-- Mỗi nhân viên chỉ có tối đa một địa điểm mặc định đang hoạt động.
CREATE UNIQUE INDEX uq_dm_dia_diem_nhan_hang_mac_dinh
ON dm_dia_diem_nhan_hang (
    nhan_vien_id
)
WHERE
    la_mac_dinh = TRUE
    AND active = TRUE;


-- Phục vụ lấy danh sách địa điểm của nhân viên.
CREATE INDEX idx_dm_dia_diem_nhan_hang_nhan_vien
ON dm_dia_diem_nhan_hang (
    nhan_vien_id,
    active,
    thu_tu_hien_thi
);


-- ============================================================
-- 6. TẠO LẠI FOREIGN KEY CỦA ĐƠN HÀNG
-- ============================================================

ALTER TABLE nv_don_hang
ADD CONSTRAINT nv_don_hang_dia_diem_nhan_id_fkey
FOREIGN KEY (
    dia_diem_nhan_id
)
REFERENCES dm_dia_diem_nhan_hang(id);


-- ============================================================
-- 7. TẠO TRIGGER UPDATED_AT
-- ============================================================

CREATE TRIGGER trg_dm_dia_diem_nhan_hang_updated_at
BEFORE UPDATE
ON dm_dia_diem_nhan_hang
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();
